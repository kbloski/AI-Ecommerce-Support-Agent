=== ORIGINAL STANDALONE GENERATOR SOURCE (prompt builder and parameters) ===

import json

from di.container import Container

from domain.models.llm.llm_message import LlmMessage
from domain.models.offer.offer_source_element import OfferSourceElement

from domain.enums.llm_message_role import LlmMessageRole
from domain.enums.offer_source_element_type import OfferSourceElementType

from application.services.llm_context_builder import build_llm_section
from application.mappers.offer_source_element_mapper import OfferSourceElementMapper


# =====================================================
# MAIN HANDLER
# =====================================================

def generate_offer_source_elements_handler(
    offer_source_id: int,
    element_types: list[OfferSourceElementType],
    examples_per_type: int = 3,
) -> list[dict]:

    container = Container()

    offer_source_service = container.offer_source_service()
    offer_source_elements_repository = container.offer_source_elements_repository()
    ai_service = container.ai_service()

    # ----------------------------
    # LOAD OFFER SOURCE CONTEXT
    # ----------------------------

    offer_source_context = offer_source_service.build_llm_context(
        offer_source_id=offer_source_id
    )

    # ----------------------------
    # GENERATE OFFER ELEMENTS
    # ----------------------------

    chat = [
        LlmMessage(
            role=LlmMessageRole.SYSTEM,
            content=get_system_prompt(),
        ),
        LlmMessage(
            role=LlmMessageRole.USER,
            content=get_offer_elements_generation_prompt(
                offer_source_context=offer_source_context,
                element_types=element_types,
                examples_per_type=examples_per_type,
            ),
        ),
    ]

    response = ai_service.chat_llm(chat).content

    json_data = json.loads(response)

    # ----------------------------
    # SAVE TO DATABASE
    # ----------------------------

    created_elements = []

    for element_data in json_data.get("offer_elements", []):

        if not isinstance(element_data, dict):
            continue

        try:
            element_type = OfferSourceElementType(element_data.get("type"))
        except (TypeError, ValueError):
            continue

        if element_type not in element_types:
            continue

        name = str(element_data.get("name", "")).strip()

        if not name:
            continue

        description = str(element_data.get("description", "")).strip() or None

        element = OfferSourceElement(
            offer_source_id=offer_source_id,
            type=element_type,
            name=name,
            description=description,
        )

        created_elements.append(
            offer_source_elements_repository.create(element)
        )

    # Group by type (stable sort by type, then name) — matches how
    # OfferSourceElementsRepository.find_for_offer_source orders results.
    created_elements.sort(
        key=lambda element: (str(element.type), element.name)
    )

    return [OfferSourceElementMapper.to_dto(element).to_dict() for element in created_elements]

# =====================================================
# OFFER ELEMENT TYPES CONTEXT
# =====================================================

OFFER_SOURCE_ELEMENT_TYPE_DEFINITIONS: dict[
    OfferSourceElementType,
    str,
] = {
    OfferSourceElementType.BENEFIT:
        "What the customer gains from the offer.",

    OfferSourceElementType.FEATURE:
        "What the offer has or does.",

    OfferSourceElementType.PROBLEM_SOLVED:
        "The problem that exists before the solution is applied.",

    OfferSourceElementType.USE_CASE:
        "A concrete usage scenario for the offer.",

    OfferSourceElementType.DIFFERENTIATOR:
        "A meaningful distinction from alternatives.",

    OfferSourceElementType.LIMITATION:
        "A meaningful restriction, boundary, or trade-off of the offer.",
}


def build_offer_element_types_context(
    element_types: list[OfferSourceElementType],
) -> str:

    definitions = [
        {
            "type": element_type.value,
            "definition": OFFER_SOURCE_ELEMENT_TYPE_DEFINITIONS[element_type],
        }
        for element_type in element_types
    ]

    return build_llm_section(
        tag="offer_element_types",
        content=json.dumps(
            definitions,
            ensure_ascii=False,
            indent=2,
        ),
    )
